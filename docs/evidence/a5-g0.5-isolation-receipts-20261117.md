export class RateLimiter {
  constructor(limit, windowMs) { this.limit = limit; this.windowMs = windowMs; this.hits = new Map(); }
  allow(key, now = Date.now()) {
    const bucket = Math.floor(now / this.windowMs);
    const prev = this.hits.get(key);

}
